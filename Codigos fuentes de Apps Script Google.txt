function obtenerDatosUsuarios() {
  const sheet = SS.getSheetByName("Usuarios");
  if (!sheet) {
    Logger.log("ERROR: La hoja 'Usuarios' no fue encontrada.");
    return [];
  }
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return []; // Retorna vacío si solo hay encabezado o no hay datos
  data.shift(); // Elimina la fila de encabezado
  return data.map(row => ({ codigo: String(row[0]).trim(), nombre: String(row[1]).trim(), cuota: String(row[2]).trim() }));
}

function agregarUsuario(userData) {
  try {
    const sheet = SS.getSheetByName("Usuarios");
    if (!sheet) throw new Error("La hoja 'Usuarios' no existe.");

    // Verifica si el código ya existe para evitar duplicados
    const existingCodes = sheet.getRange("A:A").getValues().flat().map(String);
    if (existingCodes.includes(String(userData.codigo).trim())) {
      return { status: "error", message: `El código ${userData.codigo} ya existe.` };
    }

    sheet.appendRow([String(userData.codigo).trim(), String(userData.nombre).trim()]);
    return { status: "success", message: "Usuario agregado correctamente." };
  } catch (e) {
    Logger.log(`Error al agregar usuario: ${e.message}`);
    return { status: "error", message: `Error en el servidor al agregar usuario: ${e.message}` };
  }
}

function editarUsuario(userData) {
  try {
    const sheet = SS.getSheetByName("Usuarios");
    if (!sheet) throw new Error("La hoja 'Usuarios' no existe.");

    const data = sheet.getDataRange().getValues();
    // Encuentra la fila del usuario por código (ignorando el encabezado)
    const rowIndex = data.findIndex((row, index) => index > 0 && String(row[0]).trim() === String(userData.codigo).trim());
    
    if (rowIndex > -1) {
      // rowIndex + 1 porque getDataRange() es 0-indexado, pero getRange() es 1-indexado
      // La columna 2 es para el nombre
      sheet.getRange(rowIndex + 1, 2).setValue(String(userData.nombre).trim());
      return { status: "success", message: "Usuario actualizado correctamente." };
    }
    return { status: "error", message: "Usuario no encontrado." };
  } catch (e) {
    Logger.log(`Error al editar usuario: ${e.message}`);
    return { status: "error", message: `Error en el servidor al editar usuario: ${e.message}` };
  }
}

function eliminarUsuario(codigo) {
  try {
    const sheet = SS.getSheetByName("Usuarios");
    if (!sheet) throw new Error("La hoja 'Usuarios' no existe.");

    const data = sheet.getDataRange().getValues();
    // Encuentra la fila del usuario por código (ignorando el encabezado)
    const rowIndex = data.findIndex((row, index) => index > 0 && String(row[0]).trim() === String(codigo).trim());

    if (rowIndex > -1) {
      // rowIndex + 1 es la fila real en la hoja
      sheet.deleteRow(rowIndex + 1);
      return { status: "success", message: "Usuario eliminado correctamente." };
    }
    return { status: "error", message: "Usuario no encontrado." };
  } catch (e) {
    Logger.log(`Error al eliminar usuario: ${e.message}`);
    return { status: "error", message: `Error en el servidor al eliminar usuario: ${e.message}` };
  }
}

function guardarBitacoraYAtender(formData) {
  try {
    const tecnicoNombre = formData.tecnico;
    const tecnicoCodigo = Object.keys(getUsuariosMap()).find(key => getUsuariosMap()[key] === tecnicoNombre);

    guardarRegistroEnBitacora(formData, tecnicoCodigo); // Registrar en Bitácora

    // Procesar según la fuente
    if (formData.source === 'reportes') {
      atenderMovilDeReportes(formData.reportId, tecnicoCodigo);
    } else if (formData.source === 'inventario') {
      // No se hace nada adicional: no se borra, no se mueve, no se actualiza fecha
      // Solo ya se guardó en Bitácora arriba
    }

    return { status: "success", message: "Bitácora guardada y móvil atendido." };
  } catch (e) {
    Logger.log(`Error en guardarBitacoraYAtender: ${e.message}`);
    return { status: "error", message: `Error en el servidor: ${e.message}` };
  }
}

function atenderMovilDeReportes(reportId, tecnicoCodigo) {
  if (!reportId || !tecnicoCodigo) {
    throw new Error("Parámetros inválidos: reportId y tecnicoCodigo son requeridos.");
  }

  const reportesSheet = SS.getSheetByName("Reportes");
  if (!reportesSheet) throw new Error("La hoja 'Reportes' no existe.");

  let historialSheet = SS.getSheetByName("Historial");
  if (!historialSheet) {
    historialSheet = SS.insertSheet("Historial");
    historialSheet.appendRow([
      "Fecha y Hora Atención", "Móvil", "Estado GPS", "Patio",
      "Fecha Reporte", "Hora Reporte", "Diagnóstico",
      "Técnico Nombre", "Técnico Código"
    ]);
  }

  const data = reportesSheet.getDataRange().getValues();
  const headers = data[0];
  const estadoColIndex = headers.indexOf("Estado");
  if (estadoColIndex === -1) throw new Error("No se encontró la columna 'Estado' en Reportes.");

  const usuariosMap = getUsuariosMap();

  // Buscar fila con el reportId y que no esté atendida
  const rowIndex = data.findIndex((row, idx) =>
    idx > 0 &&
    String(row[6]).trim() === String(reportId).trim() &&
    String(row[estadoColIndex]).toLowerCase() !== "atendido"
  );

  if (rowIndex === -1) {
    throw new Error("El reporte no fue encontrado o ya fue atendido.");
  }

  const reportRowData = data[rowIndex];
  const nombreTecnico = usuariosMap[String(tecnicoCodigo).trim()] || "Desconocido";

  // Calcular fila real (porque rowIndex incluye el encabezado como índice 0)
  const filaReal = rowIndex + 1;

  // 1. Marcar como atendido antes de mover
  reportesSheet.getRange(filaReal, estadoColIndex + 1).setValue("Atendido");

  // 2. Construir entrada para Historial
  const historialEntry = [
    new Date(),                          // Fecha y Hora Atención
    String(reportRowData[0] || '').trim(), // Móvil
    String(reportRowData[1] || '').trim(), // Estado GPS
    String(reportRowData[2] || '').trim(), // Patio
    reportRowData[3],                    // Fecha Reporte
    reportRowData[4],                    // Hora Reporte
    String(reportRowData[5] || '').trim(), // Diagnóstico
    nombreTecnico,                       // Técnico Nombre
    String(tecnicoCodigo || '').trim()   // Técnico Código
  ];

  // 3. Insertar en Historial
  historialSheet.appendRow(historialEntry);

  // 4. Borrar la fila original en Reportes
  reportesSheet.deleteRow(filaReal);
}

function obtenerDatosDeReportes() {
  const sheet = SS.getSheetByName("Reportes");
  if (!sheet) {
    Logger.log("ERROR: La hoja 'Reportes' no fue encontrada.");
    return [];
  }
  // Asume que los datos comienzan en la fila 3 (después de 2 filas de encabezado)
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();

  // Si no hay encabezados o datos después de la fila 2, retorna vacío
  if (lastRow < 3 || lastCol === 0) return []; 

  const dataRange = sheet.getRange(3, 1, lastRow - 2, lastCol);
  const data = dataRange.getValues();
  const reportIdColIndex = 6; // Columna G (0-indexed) para ReportID

  // Iterate and ensure each row has a UUID. If not, generate and set it.
  const updatedData = data.map((row, i) => {
    let reportId = String(row[reportIdColIndex] || '').trim();
    if (!reportId) {
      reportId = Utilities.getUuid(); // Generate UUID
      // Update the sheet directly for the new UUID
      sheet.getRange(i + 3, reportIdColIndex + 1).setValue(reportId); // i+3 for 1-indexed row, reportIdColIndex+1 for 1-indexed column
    }
    // Return row data including the (potentially new) reportId
    // Ensure the array has enough elements to include the ReportID column, even if it was empty initially.
    const fullRow = [...row];
    fullRow[reportIdColIndex] = reportId; // Place the reportId at the correct index
    
    // Convert all relevant cells to string for consistency in client-side filtering/display
    return fullRow.map(cell => String(cell || '').trim()); 
  });
  
  // Return the processed data. The client-side will now expect 7 columns (0-6) + ReportID
  // We'll pass the ReportID as a separate item at the end of each row array
  // so client can access it easily.
  return updatedData.map(row => [...row.slice(0, 6), row[reportIdColIndex]]); // Return columns A-F + ReportID
}

function obtenerDatosDeInventario() {
  const sheet = SS.getSheetByName("Inventario");
  if (!sheet) {
    Logger.log("ERROR: La hoja 'Inventario' no fue encontrada.");
    return [];
  }
  // Asume que los datos del inventario comienzan en la fila 2 (después de 1 fila de encabezado)
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();

  if (lastRow < 3 || lastCol === 0) return []; // No hay datos válidos

  // Rango de datos: desde la fila 2 hasta la última fila, todas las columnas
  const data = sheet.getRange(3, 1, lastRow - 1, lastCol).getValues();
  
  // Retorna los datos como cadenas, formateando las fechas
  return data.map(row => {
    // Patio, Móvil, Estado, Última Atención (index 3), Fecha Actual (index 4), Diagnóstico
    const ultimaAtencion = row[4];
    const fechaActual = row[5];

    // Formatear las fechas a 'dd/mm/yyyy' si son objetos Date
    const formattedUltimaAtencion = (ultimaAtencion instanceof Date) ? Utilities.formatDate(ultimaAtencion, TIME_ZONE, "dd/MM/yyyy") : String(ultimaAtencion || '').trim();
    const formattedFechaActual = (fechaActual instanceof Date) ? Utilities.formatDate(fechaActual, TIME_ZONE, "dd/MM/yyyy") : String(fechaActual || '').trim();

    return [
      String(row[0] || '').trim(), // Patio
      String(row[1] || '').trim(), // Móvil
      String(row[2] || '').trim(), // Hora
      String(row[3] || '').trim(), // Estado
      formattedUltimaAtencion,     // Última Atención (formateada)
      formattedFechaActual,        // Fecha Actual (formateada)
      String(row[6] || '').trim()  // Diagnóstico
    ];
  });
}

function obtenerDatosDeResumen(filtros) {
  const resumenSheet = SS.getSheetByName("Resumen");
  if (!resumenSheet) return { error: "Hoja Resumen no encontrada." };

  let fechaInicioParaHoja = null;
  if (filtros.fechaInicio) {
    fechaInicioParaHoja = new Date(filtros.fechaInicio + 'T12:00:00'); 
  }

  // Escribir los valores de los filtros en las celdas correspondientes.
  resumenSheet.getRange("A1").setValue(fechaInicioParaHoja);

  // 2. Dar tiempo a que las fórmulas QUERY se recalcuen
  SpreadsheetApp.flush(); // Fuerza a Apps Script a aplicar los cambios y recalcular fórmulas

  // 3. Leer los resultados de las tablas generadas por QUERY
  const tiposMantenimiento = resumenSheet.getRange("A21:H25").getValues().filter(r => r[0] !== '');
  const atencionesPorPatio = resumenSheet.getRange("A27:H33").getValues().filter(r => r[0] !== '');
  const atencionesPorTecnico = resumenSheet.getRange("A3:H19").getValues().filter(r => r[0] !== '');
  const estadoModem = resumenSheet.getRange("A35:H42").getValues().filter(r => r[0] !== '');

  return {
    tiposMantenimiento: tiposMantenimiento,
    atencionesPorPatio: atencionesPorPatio,
    atencionesPorTecnico: atencionesPorTecnico,
    estadoModem: estadoModem
  };
}

function getResumenEquiposTablesData() {
  const resumenEquiposSheet = SS.getSheetByName("Resumen de equipos");
  if (!resumenEquiposSheet) {
    console.error("La hoja 'Resumen de equipos' no se encontró.");
    return {
      estadoFlota: [],
      bocinasDañadas: [],
      estadoModem: [],
      estadoBocinas: []
    };
  }

  // Definir los rangos de cada tabla
  const ranges = {
    estadoFlota: "K4:N11",
    bocinasDañadas: "K14:N21",
    estadoModem: "K24:N28",
    estadoBocinas: "K31:N37"
  };

  const tablesData = {};

  for (const tableName in ranges) {
    if (ranges.hasOwnProperty(tableName)) {
      const range = ranges[tableName];
      try {
        tablesData[tableName] = resumenEquiposSheet.getRange(range).getValues();
      } catch (e) {
        console.error(`Error al obtener datos para la tabla ${tableName} en el rango ${range}: ${e.message}`);
        tablesData[tableName] = []; // Asegurarse de que el array esté vacío en caso de error
      }
    }
  }
  return tablesData;
}

function getUsuariosMap() {

    return {};
  }
  const data = sheet.getDataRange().getValues();
  const map = {};
  // CORRECCIÓN: map[codigo] = nombre. Antes estaba map[codigo] = codigo.
  data.slice(1).forEach(row => { // Omite el encabezado
    if (row[0] && row[1]) {

function getContador(tecnicoCodigo) {
  const repuestasSheet = SpreadsheetApp.getActive().getSheetByName("RepuestasForms");
  if (!repuestasSheet) return 0;

  const now = new Date();
  const effectiveDate = new Date(now);
  if (now.getHours() < 8) {
    effectiveDate.setDate(effectiveDate.getDate() - 1);
  }
  effectiveDate.setHours(0, 0, 0, 0);

  const TIME_ZONE = Session.getScriptTimeZone();
  const todayStr = Utilities.formatDate(effectiveDate, TIME_ZONE, "yyyy-MM-dd");

  const data = repuestasSheet.getDataRange().getValues();
  let count = 0;

  for (let i = 1; i < data.length; i++) {
    const timestamp = data[i][0];      // Columna A: Fecha de registro
    const codigoTecnico = data[i][18]; // Columna Q: Código del técnico (índice 16)

    if (!(timestamp instanceof Date)) continue;
    if (String(codigoTecnico).trim() !== String(tecnicoCodigo).trim()) continue;

    const registroDate = new Date(timestamp);
    if (registroDate.getHours() < 8) {
      registroDate.setDate(registroDate.getDate() - 1);
    }
    registroDate.setHours(0, 0, 0, 0);

    const registroStr = Utilities.formatDate(registroDate, TIME_ZONE, "yyyy-MM-dd");

    if (registroStr === todayStr) {
      count++;
    }
  }

  return count;
}


function getContadorUsuarioActual(codigo) {

}

function getCuotaDiaria(codigoTecnico) {
  const sheet = SS.getSheetByName("Usuarios");
  if (!sheet) {
    Logger.log("ERROR: La hoja 'Usuarios' no fue encontrada.");
    return 0; // Devolver 0 si la hoja no existe
  }

  const data = sheet.getDataRange().getValues();
  // !!! AJUSTA ESTOS ÍNDICES SEGÚN TU HOJA "USUARIOS" !!!
  const CODIGO_COL = 0; // Columna A (código del técnico)
  const CUOTA_COL = 2;  // Columna C (cuota diaria)

  for (let i = 1; i < data.length; i++) { // Empieza en 1 para saltar la fila de encabezado
    const userCodigo = String(data[i][CODIGO_COL]).trim();
    if (userCodigo === String(codigoTecnico).trim()) {
      // Asegúrate de que el valor en la hoja sea numérico. Si no lo es, se convertirá a 0.
      return Number(data[i][CUOTA_COL]) || 0;
    }
  }
  return 0; // Devolver 0 si el técnico no se encuentra
}

    function filterTable(type) {
        const input = document.getElementById(`filterInput${type === 'reportes' ? 'Reportes' : 'Inventario'}`);
        const filterText = input.value.toUpperCase();
        const table = document.getElementById(`${type}Table`);
        const tr = table.getElementsByTagName("tr");

        const patioFilterElement = document.getElementById(`patioFilter${type === 'reportes' ? 'Reportes' : 'Inventario'}`);
        const selectedPatio = patioFilterElement ? patioFilterElement.value.toUpperCase() : '';

        for (let i = 1; i < tr.length; i++) { // Start from 1 to skip header row
            let displayRow = false;
            let patioMatch = true;
            let textMatch = false;

            const cells = tr[i].getElementsByTagName("td");
            // Check for text match across all cells
            for (let j = 0; j < cells.length; j++) {
                const cellText = cells[j].textContent || cells[j].innerText;
                if (cellText.toUpperCase().indexOf(filterText) > -1) {
                    textMatch = true;
                    break;
                }
            }

            // Check for patio match
            if (selectedPatio) {
                // Adjust index based on table type
                const patioCellIndex = (type === 'reportes' ? 2 : 0); // Patio is at index 2 for reports, 0 for inventory
                if (cells[patioCellIndex]) {
                    const cellPatio = cells[patioCellIndex].textContent.toUpperCase();
                    if (cellPatio.indexOf(selectedPatio) === -1) {
                        patioMatch = false;
                    }
                } else {
                    patioMatch = false; // If patio cell doesn't exist, don't match
                }
            }

            if (textMatch && patioMatch) {
                tr[i].style.display = "";
            } else {
                tr[i].style.display = "none";
            }
        }
    }

            headerRowHtml += `<th class="${isTotalColumnHeader ? 'total-header' : ''}">${header}</th>`;

    function drawPieChart(element, title, data) {
        const dataTable = google.visualization.arrayToDataTable(data);
        const options = { title: title, pieHole: 0.4, backgroundColor: 'transparent' };
        const chart = new google.visualization.PieChart(element);
        chart.draw(dataTable, options);
    }

    function drawBarChart(element, title, data) {
        const dataTable = google.visualization.arrayToDataTable(data);
        const options = { title: title, legend: { position: 'none' }, backgroundColor: 'transparent' };
        const chart = new google.visualization.BarChart(element);
        chart.draw(dataTable, options);
    }

    // --- USUARIOS SECTION ---
    function renderUsuarios(users) {
        const tableBody = document.getElementById('usuariosTable').querySelector('tbody');
        tableBody.innerHTML = '';
        users.forEach(user => {
            const tr = tableBody.insertRow();
            tr.insertCell().textContent = user.codigo;
            tr.insertCell().textContent = user.nombre;
            const actionTd = tr.insertCell();
            
            const editBtn = document.createElement('button');
            editBtn.textContent = 'Editar';
            editBtn.onclick = () => showUsuarioForm(user);
            actionTd.appendChild(editBtn);

            const deleteBtn = document.createElement('button');
            deleteBtn.textContent = 'Eliminar';
            deleteBtn.className = 'btn-danger';
            deleteBtn.style.marginLeft = '10px';
            deleteBtn.onclick = () => deleteUsuario(user.codigo);
            actionTd.appendChild(deleteBtn);
        });
    }

    function showUsuarioForm(user) {
        const formContainer = document.getElementById('usuarioFormContainer');
        const form = document.getElementById('usuarioForm');
        form.reset();

        if (user) { // Edit mode
            document.getElementById('usuarioFormTitle').textContent = 'Editar Usuario';
            document.getElementById('usuarioCodigo').value = user.codigo;
            document.getElementById('usuarioCodigo').disabled = true; // Code remains disabled for editing
            document.getElementById('usuarioNombre').value = user.nombre;
        } else { // Add mode
            document.getElementById('usuarioFormTitle').textContent = 'Agregar Usuario';
            document.getElementById('usuarioCodigo').disabled = false;
            document.getElementById('usuarioCodigo').value = ''; // Clear for new user
            document.getElementById('usuarioNombre').value = ''; // Clear for new user
        }
        formContainer.classList.remove('hidden');
    }

    function hideUsuarioForm() {
        document.getElementById('usuarioFormContainer').classList.add('hidden');
    }

    function handleUsuarioSubmit(event) {
        event.preventDefault();
        const userData = {
            codigo: document.getElementById('usuarioCodigo').value.trim(),
            nombre: document.getElementById('usuarioNombre').value.trim()
        };
        
        const isEditMode = document.getElementById('usuarioCodigo').disabled;
        const serverFunction = isEditMode ? 'editarUsuario' : 'agregarUsuario';

        showLoadingOverlay();
        google.script.run.withSuccessHandler(response => {
            hideLoadingOverlay();
            showToast(response.message);
            if(response.status === 'success') {
                hideUsuarioForm();
                loadData('usuarios');
            }
        })[serverFunction](userData);
    }
    
    function deleteUsuario(codigo) {
        if (confirm(`¿Está seguro de que desea eliminar al usuario con código ${codigo}?`)) {
            showLoadingOverlay();
            google.script.run.withSuccessHandler(response => {
                hideLoadingOverlay();
                showToast(response.message);
                if (response.status === 'success') {
                    loadData('usuarios');
                }
            }).eliminarUsuario(codigo);
        }
    }

        const toast = document.getElementById("toast");
        toast.textContent = message;
        toast.className = "show";
        setTimeout(() => { toast.className = toast.className.replace("show", ""); }, 3000);
    }
